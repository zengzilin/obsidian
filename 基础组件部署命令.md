# 基础组件部署命令

# 获取argocd登录密码：
kubectl -n argocd get secret argocd-initial-admin-secret -o jsonpath="{.data.password}" | base64 -d; echo



# helm部署 sftp 
<font style="color:black;background-color:#FFFFFF;">helm repo add emberstack https://emberstack.github.io/helm-charts</font>



helm install sftp emberstack/sftp --version 5.1.69 -f values.yaml -n test2-ksa-middleware



# helm部署 promethues
helm upgrade prometheus prometheus-community/kube-prometheus-stack --version 54.0.1 -f  ./values.yaml -n monitoring



# helm部署 rabbitmq
helm install rabbitmq-ksa2  bitnami/rabbitmq --version 11.11 -f  ./values-ksa2.yaml  -n ksa2-middleware



# helm 更新 rabbitmq
helm upgrade rabbitmq-dev-tmp  bitnami/rabbitmq --version 11.11 -f  ./values-dev.yaml  -n middleware



# rabbitmq报错 wal table报错
由于rabbit mq用NFS存储导致的文件损坏

更改启动方式为并行启动

helm upgrade rabbitmq-mtls bitnami/rabbitmq --version 11.11 -n middleware \

  --set podManagementPolicy=Parallel -f values_mtls.yaml 

# helm部署redis
helm install redis-cluster bitnami/redis-cluster  --version=8.3.8  -f values.yaml  -n middleware



# helm更新redis
export REDIS_PASSWORD=$(kubectl get secret --namespace "middleware" redis-cluster -o jsonpath="{.data.redis-password}" | base64 -d)



helm upgrade redis-cluster bitnami/redis-cluster  --version=8.3.8 -f  ./values.yaml -n middleware





# helm 重装 redis
helm uninstall redis-cluster

kubectl delete pvc redis-data-redis-cluster-0 redis-data-redis-cluster-1 redis-data-redis-cluster-2 redis-data-redis-cluster-3 redis-data-redis-cluster-4 redis-data-redis-cluster-5



helm install redis-cluster bitnami/redis-cluster  --version=8.3.8  -f values.yaml  -n middleware





# helm 安装 k8tz
helm repo add k8tz [https://k8tz.github.io/k8tz/](https://k8tz.github.io/k8tz/)

helm install k8tz k8tz/k8tz --set timezone=Asia/Riyadh -n middleware
